Circuit Wiring Schematic

Connect VCC pin to 5V/3.3V power rail, GND pin to common ground. Connect signal output pin to GPIO #2 on the Arduino microcontroller.

💡 Ensure microcontrollers are completely powered down before connecting GPIO jumper wires to prevent short circuits.

Project Code (CPP)

// JR Learners Project #1 - Smart Plant Automatic Watering System
// Microcontroller Code for Arduino

const int SENSOR_PIN = 2;
const int STATUS_LED = 13;

void setup() {
  pinMode(SENSOR_PIN, INPUT);
  pinMode(STATUS_LED, OUTPUT);
  Serial.begin(9600);
  Serial.println("Starting Project #1: Smart Plant Automatic Watering System");
}

void loop() {
  int sensorState = digitalRead(SENSOR_PIN);
  if (sensorState == HIGH) {
    digitalWrite(STATUS_LED, HIGH);
    Serial.println("Sensor Triggered!");
  } else {
    digitalWrite(STATUS_LED, LOW);
  }
  delay(100);
}
